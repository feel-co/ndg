1//! Nix file extraction logic for finding attribute bindings with doc comments.
2//!
3//! This module walks a parsed `nixon` syntax tree to find all attribute
4//! path-value bindings that are immediately preceded by a `/** ... */`
5//! (Nixdoc-style) comment. Nested attribute sets are traversed recursively
6//! so that deeply-nested paths (e.g. `lib.strings.concatStrings`) are
7//! reported with their full dotted path.
8use std::path::Path;
9
10use nixon::{
11  Element,
12  Node,
13  SyntaxKind,
14  ast::{self, AstNode},
15};
16
17use crate::types::{Location, RawEntry};

19/// Pre-computed line start positions for O(log n) line number lookups.
20struct LineIndex {
21  starts: Vec<usize>,
22}
23
24impl LineIndex {
25  fn new(src: &str) -> Self {
26    let mut starts = vec![0];
27    for (byte_offset, c) in src.char_indices() {
28      if c == '\n' {
29        starts.push(byte_offset + 1);
30      }
31    }
32    Self { starts }
33  }
34
35  fn line_of_offset(&self, byte_offset: usize) -> u32 {
36    let offset = byte_offset.min(self.starts.last().copied().unwrap_or(0));
37    match self.starts.binary_search(&offset) {
38      Ok(line) => u32::try_from(line + 1).unwrap_or(u32::MAX),
39      Err(insert_idx) => u32::try_from(insert_idx).unwrap_or(u32::MAX),
40    }
41  }
42}

44/// Extract all raw doc-comment entries from a Nix source string.
45///
46/// `file_path` is only used to populate [`Location`] metadata; the actual
47/// parsing is done on `src`.
48///
49/// Parse errors from `nixon` are silently dropped at the node level: because
50/// `nixon` is an error-tolerant parser it always returns a tree, but
51/// individual nodes may be `None` when their structure could not be
52/// determined. Those nodes are skipped.
53pub fn extract_entries(src: &str, file_path: &Path) -> Vec<RawEntry> {
54  let Ok(document) = nixon::parse_syntax(src) else {
55    return Vec::new();
56  };
57  let Some(root) = ast::Root::cast(document.root()) else {
58    return Vec::new();
59  };
60  let line_index = LineIndex::new(src);
61
62  let mut entries = Vec::new();
63
64  // nixon::Root wraps a single body expression. In flat assignment files (most
65  // nixpkgs-style files) the body is itself an AttrSet; in others it may be a
66  // `let … in` expression.  Rather than hardcoding the root as "has entries",
67  // we recurse into the root's expression tree directly.
68  if let Some(body) = root.expression() {
69    collect_from_expr(body.syntax(), &[], file_path, &line_index, &mut entries);
70  }
71
72  entries
73}

75/// Recursively walk an expression node, collecting doc-commented bindings.
76///
77/// Only attribute-set and let-in nodes can directly contain attribute
78/// entries. For any other node we descend into all children looking for nested
79/// attrsets (e.g. the body of a lambda or a conditional).
80///
81/// `path_prefix` holds the attribute path segments accumulated from ancestor
82/// attrsets so that nested entries are reported with their full dotted path.
83fn collect_from_expr(
84  node: Node<'_, '_>,
85  path_prefix: &[String],
86  file_path: &Path,
87  line_index: &LineIndex,
88  out: &mut Vec<RawEntry>,
89) {
90  use SyntaxKind::{AttributeSet, LetIn};
91
92  match node.kind() {
93    AttributeSet => {
94      if let Some(attrset) = ast::AttributeSet::cast(node) {
95        collect_entries(
96          attrset.syntax(),
97          path_prefix,
98          file_path,
99          line_index,
100          out,
101        );
102      }
103    },
104    LetIn => {
105      if let Some(let_in) = ast::LetIn::cast(node) {
106        collect_entries(
107          let_in.syntax(),
108          path_prefix,
109          file_path,
110          line_index,
111          out,
112        );
113        // Also recurse into the body expression (the part after `in`), since
114        // it may be an attrset containing doc-commented bindings.
115        if let Some(body) = let_in.body() {
116          collect_from_expr(
117            body.syntax(),
118            path_prefix,
119            file_path,
120            line_index,
121            out,
122          );
123        }
124      }
125    },
126    _ => {
127      // Not an entry container, so we descend into all children looking for
128      // nested attribute sets (e.g. the body of a lambda or a with
129      // expression).
130      for child in node.child_nodes() {
131        collect_from_expr(child, path_prefix, file_path, line_index, out);
132      }
133    },
134  }
135}

137/// Walk one attrset/let-in level, extract doc-commented bindings, and recurse
138/// into nested attrsets.
139fn collect_entries(
140  container: Node<'_, '_>,
141  path_prefix: &[String],
142  file_path: &Path,
143  line_index: &LineIndex,
144  out: &mut Vec<RawEntry>,
145) {
146  for element in container.children() {
147    let Element::Node(node) = element else {
148      continue;
149    };
150
151    let Some(binding) = ast::Binding::cast(node) else {
152      continue;
153    };
154
155    let Some(attrpath) = binding.path() else {
156      continue;
157    };
158    let components: Vec<ast::AttributeComponent<'_, '_>> =
159      attrpath.components().collect();
160    let segments: Vec<String> = components
161      .iter()
162      .copied()
163      .filter_map(attribute_to_string)
164      .collect();
165
166    // Skip if any segment was dynamic (could not be statically resolved);
167    // a partial path like ["foo", "bar"] for `foo.${expr}.bar` would be wrong.
168    if segments.len() != components.len() || segments.is_empty() {
169      continue;
170    }
171
172    let comment = leading_doc_comment(attrpath.syntax());
173
174    // Only build full_path if we have a doc comment or need to recurse
175    if comment.is_some() || binding.value().is_some() {
176      let full_path: Vec<String> =
177        path_prefix.iter().chain(segments.iter()).cloned().collect();
178
179      if let Some(comment) = comment {
180        let start = components
181          .first()
182          .copied()
183          .map(attribute_start)
184          .unwrap_or_default();
185        let location = Location {
186          file: file_path.to_path_buf(),
187          line: line_index.line_of_offset(start),
188        };
189
190        out.push(RawEntry {
191          attr_path: full_path.clone(),
192          comment: comment.to_owned(),
193          location,
194        });
195      }
196
197      // recurse into RHS if it is an attrset
198      if let Some(value) = binding.value() {
199        collect_from_expr(
200          value.syntax(),
201          &full_path,
202          file_path,
203          line_index,
204          out,
205        );
206      }
207    }
208  }
209}

211/// Find a doc comment among the trivia Nixon attaches to an attribute path.
212fn leading_doc_comment<'src>(attrpath: Node<'_, 'src>) -> Option<&'src str> {
213  let mut comment = None;
214  scan_leading_trivia(attrpath, &mut comment);
215  comment
216}
217
218fn scan_leading_trivia<'src>(
219  node: Node<'_, 'src>,
220  comment: &mut Option<&'src str>,
221) {
222  for element in node.children() {
223    match element.kind() {
224      SyntaxKind::Whitespace => {},
225      SyntaxKind::DocComment => *comment = Some(element.text()),
226      SyntaxKind::LineComment | SyntaxKind::BlockComment => *comment = None,
227      _ => {
228        if let Element::Node(child) = element {
229          scan_leading_trivia(child, comment);
230        }
231        return;
232      },
233    }
234  }
235}

237fn attribute_start(attribute: ast::AttributeComponent<'_, '_>) -> usize {
238  match attribute {
239    ast::AttributeComponent::Identifier(identifier) => {
240      identifier.range().start().as_usize()
241    },
242    ast::AttributeComponent::String(string) => {
243      first_non_trivia_start(string.syntax())
244    },
245    ast::AttributeComponent::Interpolation(interpolation) => {
246      first_non_trivia_start(interpolation.syntax())
247    },
248  }
249}
250
251fn first_non_trivia_start(node: Node<'_, '_>) -> usize {
252  for element in node.children() {
253    match element {
254      Element::Token(token) if token.kind().is_trivia() => {},
255      Element::Token(token) => return token.range().start().as_usize(),
256      Element::Node(child) => return first_non_trivia_start(child),
257    }
258  }
259  node.range().start().as_usize()
260}
261
262fn first_non_trivia_kind(node: Node<'_, '_>) -> Option<SyntaxKind> {
263  for element in node.children() {
264    match element {
265      Element::Token(token) if token.kind().is_trivia() => {},
266      Element::Token(token) => return Some(token.kind()),
267      Element::Node(child) => return first_non_trivia_kind(child),
268    }
269  }
270  None
271}

273/// Convert an [`ast::AttributeComponent`] into a plain `String` segment.
274///
275/// Handles simple identifiers and string-literal keys. Dynamic keys
276/// (e.g. `${ expr }`) cannot be statically resolved and are skipped
277/// (returning `None`).
278fn attribute_to_string(
279  attribute: ast::AttributeComponent<'_, '_>,
280) -> Option<String> {
281  match attribute {
282    ast::AttributeComponent::Identifier(identifier) => {
283      Some(identifier.text().to_owned())
284    },
285    ast::AttributeComponent::String(string) => normalize_static_string(string),
286    ast::AttributeComponent::Interpolation(_) => None,
287  }
288}
289
290fn normalize_static_string(
291  string: ast::StringExpression<'_, '_>,
292) -> Option<String> {
293  let mut contents = String::new();
294  for part in string.parts() {
295    match part {
296      ast::StringPart::Fragment(fragment) => contents.push_str(fragment.text()),
297      ast::StringPart::Interpolation(_) => return None,
298    }
299  }
300
301  let multiline =
302    first_non_trivia_kind(string.syntax()) == Some(SyntaxKind::IndentedQuote);
303  if multiline {
304    contents = strip_indented_string_whitespace(&contents);
305  }
306  Some(unescape_string(&contents, multiline))
307}

309// ponytail: remove these two helpers once Nixon exposes normalized static
310// string contents.
311fn strip_indented_string_whitespace(input: &str) -> String {
312  let input = input
313    .find('\n')
314    .filter(|&newline| {
315      input[..newline].chars().all(|character| character == ' ')
316    })
317    .map_or(input, |newline| &input[newline + 1..]);
318
319  let min_indent = input
320    .lines()
321    .filter(|line| line.chars().any(|character| character != ' '))
322    .map(|line| {
323      line
324        .chars()
325        .take_while(|&character| character == ' ')
326        .count()
327    })
328    .min()
329    .unwrap_or(usize::MAX);
330
331  let mut normalized = String::with_capacity(input.len());
332  for segment in input.split_inclusive('\n') {
333    let (line, newline) = segment
334      .strip_suffix('\n')
335      .map_or((segment, ""), |line| (line, "\n"));
336    let drop = line
337      .chars()
338      .take_while(|&character| character == ' ')
339      .count()
340      .min(min_indent);
341    normalized.extend(line.chars().skip(drop));
342    normalized.push_str(newline);
343  }
344
345  if let Some(newline) = normalized.rfind('\n')
346    && normalized[newline + 1..]
347      .chars()
348      .all(|character| character == ' ')
349  {
350    normalized.truncate(newline + 1);
351  }
352
353  normalized
354}

356fn unescape_string(input: &str, multiline: bool) -> String {
357  let mut output = String::with_capacity(input.len());
358  let mut characters = input.chars().peekable();
359
360  while let Some(character) = characters.next() {
361    match character {
362      '\\' if !multiline => push_escape(&mut output, characters.next()),
363      '\'' if multiline && characters.next_if_eq(&'\'').is_some() => {
364        match characters.next() {
365          Some('\'') | None => output.push_str("''"),
366          Some('$') => output.push('$'),
367          Some('\\') => push_escape(&mut output, characters.next()),
368          Some(character) => {
369            output.push('\'');
370            output.push('\'');
371            output.push(character);
372          },
373        }
374      },
375      character => output.push(character),
376    }
377  }
378
379  output
380}
381
382fn push_escape(output: &mut String, escaped: Option<char>) {
383  match escaped {
384    Some('n') => output.push('\n'),
385    Some('r') => output.push('\r'),
386    Some('t') => output.push('\t'),
387    Some(character) => output.push(character),
388    None => {},
389  }
390}

402  #[test]
403  fn test_extract_top_level_doc_comment() {
404    let src = r"{
405/**
406  A top-level function.
407
408  # Arguments
409
410  - [x] The input value.
411*/
412identity = x: x;
413}";
414    let entries = extract_entries(src, &path());
415    assert_eq!(entries.len(), 1);
416    assert_eq!(entries[0].attr_path, vec!["identity"]);
417    assert!(entries[0].comment.contains("A top-level function"));
418  }
