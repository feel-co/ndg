16pub fn expand(input: &DeriveInput) -> syn::Result<TokenStream> {
17  let Data::Struct(data) = &input.data else {
18    return Err(syn::Error::new_spanned(
19      input,
20      "ConfigTemplate requires a struct",
21    ));
22  };
23  let Fields::Named(fields) = &data.fields else {
24    return Err(syn::Error::new_spanned(
25      input,
26      "ConfigTemplate requires named fields",
27    ));
28  };
29  let mut values = Vec::new();
30  let mut sections = Vec::new();
31
32  for field in &fields.named {
33    let Some(name) = &field.ident else { continue };
34    let mut key = name.to_string();
35    let mut skip = false;
36    let mut nested = false;
37    let mut example: Option<Expr> = None;
38    let mut docs = Vec::new();
39    for attr in &field.attrs {
40      if attr.path().is_ident("doc") {
41        if let Meta::NameValue(meta) = &attr.meta
42          && let Expr::Lit(expr) = &meta.value
43          && let syn::Lit::Str(text) = &expr.lit
44        {
45          docs.push(text.value().trim().to_string());
46        }
47      } else if attr.path().is_ident("template") {
48        attr.parse_nested_meta(|meta| {
49          if meta.path.is_ident("nested") {
50            nested = true;
51          } else if meta.path.is_ident("example") {
52            example = Some(meta.value()?.parse()?);
53          } else {
54            return Err(meta.error("expected nested or example"));
55          }
56          Ok(())
57        })?;
58      } else if attr.path().is_ident("serde") || attr.path().is_ident("config")
59      {
60        let attributes = attr
61          .parse_args_with(Punctuated::<Meta, Token![,]>::parse_terminated)?;
62        for meta in attributes {
63          if attr.path().is_ident("serde") {
64            if meta.path().is_ident("skip")
65              || meta.path().is_ident("skip_serializing")
66            {
67              skip = true;
68            } else if meta.path().is_ident("rename") {
69              key = renamed_key(&meta)?;
70            }
71          } else if meta.path().is_ident("nested") {
72            nested = true;
73          }
74        }
75      }
76    }
77    if skip {
78      continue;
79    }
80    let docs = docs.join("\n");
81    let ty = &field.ty;
82    let optional = type_is(ty, "Option");
83    let array = nested && type_is(ty, "Vec");
84    let value_ty = if optional || array {
85      first_type_arg(ty)
86        .ok_or_else(|| syn::Error::new_spanned(ty, "expected a value type"))?
87    } else {
88      ty
89    };
90    let fallback = example.as_ref().map_or_else(
91      || quote!(<#value_ty as Default>::default()),
92      |expr| quote!(#expr),
93    );
94    let method = if nested {
95      quote!(section)
96    } else {
97      quote!(field)
98    };
99    let extra = nested.then(|| quote!(, #array));
100    let handler = if optional {
101      quote! {
102        if let Some(value) = &self.#name {
103          writer.#method(#key, #docs, value, false #extra)?;
104        } else {
105          writer.#method(#key, #docs, &#fallback, true #extra)?;
106        }
107      }
108    } else if array {
109      quote! {
110        if self.#name.is_empty() {
111          writer.section(#key, #docs, &#fallback, true, true)?;
112        } else {
113          for value in &self.#name {
114            writer.section(#key, #docs, value, false, true)?;
115          }
116        }
117      }
118    } else if !nested && type_is(ty, "Vec") && example.is_some() {
119      quote! {
120        if self.#name.is_empty() {
121          writer.field(#key, #docs, &#fallback, true)?;
122        } else {
123          writer.field(#key, #docs, &self.#name, false)?;
124        }
125      }
126    } else {
127      quote! { writer.#method(#key, #docs, &self.#name, false #extra)?; }
128    };
129    if nested {
130      sections.push(handler);
131    } else {
132      values.push(handler);
133    }
134  }
135
136  let name = &input.ident;
137  let (impl_generics, ty_generics, where_clause) =
138    input.generics.split_for_impl();
139  Ok(quote! {
140    impl #impl_generics crate::templates::ConfigTemplate for #name #ty_generics #where_clause {
141      fn write_template(&self, writer: &mut crate::templates::TemplateWriter) -> Result<(), crate::templates::TemplateError> {
142        #(#values)*
143        #(#sections)*
144        Ok(())
145      }
146    }
147  })
148}

150fn renamed_key(meta: &Meta) -> syn::Result<String> {
151  let Meta::NameValue(rename) = meta else {
152    return Err(syn::Error::new_spanned(meta, "expected rename = \"key\""));
153  };
154  let value = &rename.value;
155  let lit: LitStr = syn::parse2(quote!(#value))?;
156  Ok(lit.value())
157}
