20/// Policy for duplicate heading anchor IDs within a single page.
21#[derive(
22  Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize,
23)]
24#[serde(rename_all = "lowercase")]
25pub enum DuplicateAnchorPolicy {
26  /// Fail the build on duplicate anchors (current behaviour).
27  #[default]
28  Error,
29  /// Log a warning and keep the duplicate IDs as-is.
30  Warn,
31  /// Make anchors unique with deterministic `-1`, `-2`, ... suffixes.
32  Deduplicate,
33}
34
35impl FromStr for DuplicateAnchorPolicy {
36  type Err = String;
37
38  fn from_str(s: &str) -> Result<Self, Self::Err> {
39    match s.to_lowercase().as_str() {
40      "error" => Ok(Self::Error),
41      "warn" => Ok(Self::Warn),
42      "deduplicate" => Ok(Self::Deduplicate),
43      _ => {
44        Err(format!(
45          "Unknown duplicate anchor policy: {s}; expected `error`, `warn`, or \
46           `deduplicate`"
47        ))
48      },
49    }
50  }
51}

59  #[test]
60  fn test_duplicate_policy_default_is_error() {

63  }
64
65  #[test]
66  fn test_duplicate_policy_toml_deserialization() {
67    let config: AnchorConfig =
68      toml::from_str("on_duplicate = \"warn\"").unwrap();
69    assert_eq!(config.on_duplicate, DuplicateAnchorPolicy::Warn);
70
71    let config: AnchorConfig =
72      toml::from_str("on_duplicate = \"deduplicate\"").unwrap();
73    assert_eq!(config.on_duplicate, DuplicateAnchorPolicy::Deduplicate);
74  }
75
76  #[test]
77  fn test_duplicate_policy_apply_override() {
78    let mut config = AnchorConfig::default();
79    config.apply_override("on_duplicate", "warn").unwrap();
80    assert_eq!(config.on_duplicate, DuplicateAnchorPolicy::Warn);
81
82    let err = config.apply_override("on_duplicate", "bogus").unwrap_err();
