11/// Join an output-relative path to an output directory.
12///
13/// # Errors
14///
15/// Returns an error when `relative` is absolute or escapes its output root.
16pub fn output_path(output_dir: &Path, relative: &Path) -> Result<PathBuf> {
17  if relative.components().any(|component| {
18    matches!(
19      component,
20      Component::ParentDir | Component::RootDir | Component::Prefix(_)
21    )
22  }) {
23    bail!(
24      "output path must be relative and must not contain parent components: {}",
25      relative.display()
26    );
27  }
28
29  Ok(output_dir.join(relative))
30}

32/// Creates a fallback index page listing available documents.
33///
34/// This is used when no `index.md` is present. It generates a simple HTML
35/// listing of all processed markdown files, using their first heading or
36/// filename as the display title.
37///
38/// # Arguments
39///
40/// * `config` - The loaded configuration for documentation generation.
41/// * `markdown_files` - The list of processed markdown file paths.
42///
43/// # Returns
44///
45/// A [`String`] containing the fallback HTML content.
46#[must_use]
47pub fn create_fallback_index(
48  config: &Config,
49  markdown_files: &[PathBuf],
50) -> String {
51  let mut content = format!(
52    "<h1>{}</h1>\n<p>This is a fallback page created by ndg.</p>",
53    config.title
54  );
55
56  // Add file listing if we have an input directory
57  if let Some(input_dir) = &config.input_dir
58    && !markdown_files.is_empty()
59  {
60    let mut file_list = String::with_capacity(markdown_files.len() * 100); // preallocate based on estimated size
61    file_list.push_str("<h2>Available Documents</h2>\n<ul>\n");
62
63    for file_path in markdown_files {
64      if let Ok(rel_path) = file_path.strip_prefix(input_dir) {
65        // Skip included files that are not generated as HTML
66        if config.included_files.contains_key(rel_path) {
67          continue;
68        }
69
70        let mut html_path = rel_path.to_path_buf();
71        html_path.set_extension("html");
72
73        // Get page title from first heading or filename
74        let page_title = extract_page_title(file_path, &html_path);
75
76        // Writing to String is infallible
77        let _ = writeln!(
78          file_list,
79          "  <li><a href=\"{}\">{}</a></li>",
80          html_escape::encode_double_quoted_attribute(
81            &html_path.to_string_lossy()
82          ),
83          html_escape::encode_text(&page_title)
84        );
85      }
86    }
87
88    file_list.push_str("</ul>");
89    content.push_str(&file_list);
90  }
91
92  content
93}
