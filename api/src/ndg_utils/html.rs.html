7/// Calculate the relative path prefix needed to reach the root from a given
8/// file path For example: "docs/subdir/file.html" would return "../"
9///              "docs/subdir/nested/file.html" would return "../../"
10#[must_use]
11pub fn calculate_root_relative_path(file_rel_path: &Path) -> String {
12  let depth = file_rel_path.components().count();
13  if depth <= 1 {
14    String::new() // file is at root level
15  } else {
16    "../".repeat(depth - 1)
17  }
18}

20/// Generate proper asset paths for templates based on file location
21#[must_use]
22pub fn generate_asset_paths(
23  file_rel_path: &Path,
24) -> FxHashMap<&'static str, String> {
25  let root_prefix = calculate_root_relative_path(file_rel_path);
26
27  let mut paths = FxHashMap::default();
28  paths.reserve(6);
29  paths.insert("stylesheet_path", format!("{root_prefix}assets/style.css"));
30  paths.insert("main_js_path", format!("{root_prefix}assets/main.js"));
31  paths.insert("search_js_path", format!("{root_prefix}assets/search.js"));
32
33  // Navigation paths
34  paths.insert("index_path", format!("{root_prefix}index.html"));
35  paths.insert("options_path", format!("{root_prefix}options.html"));
36  paths.insert("search_path", format!("{root_prefix}search.html"));
37
38  paths
39}

41/// Process content through the markdown pipeline and extract plain text.
42///
43/// Converts markdown to plain text while preserving document structure:
44///
45/// - Headings and paragraphs are separated by newlines
46/// - Multiple consecutive blank lines are collapsed to single blank lines
47/// - Whitespace within lines is normalized to single spaces
48/// - Anchor markers like {#id} are removed
49#[must_use]
50pub fn content_to_plaintext(content: &str) -> String {
51  normalize_plaintext(&strip_markdown(content), true, true)
52}

54/// Extract plain text from rendered HTML.
55///
56/// Uses the same output format as [`content_to_plaintext`], but without blank
57/// lines. Block elements start a new line. The text of `<script>`, `<style>`
58/// and `<template>` elements is not included. The text of code blocks is
59/// included, so that a search can find commands and file names.
60#[must_use]
61pub fn html_to_plaintext(html: &str) -> String {
62  fn walk(node: &NodeRef, out: &mut String) {
63    if let Some(text) = node.as_text() {
64      out.push_str(&text.borrow());
65      return;
66    }
67
68    let name = node.as_element().map(|element| &*element.name.local);
69    if matches!(name, Some("script" | "style" | "template")) {
70      return;
71    }
72    let is_block = name.is_some_and(is_block_element);
73    if name == Some("br") || is_block {
74      out.push('\n');
75    }
76    for child in node.children() {
77      walk(&child, out);
78    }
79    if is_block {
80      out.push('\n');
81    }
82  }
83
84  let mut text = String::with_capacity(html.len() / 2);
85  walk(&kuchikikiki::parse_html().one(html), &mut text);
86  normalize_plaintext(&text, false, false)
87}

128/// Normalize extracted text:
129///
130/// - Multiple consecutive blank lines are collapsed to single blank lines, or
131///   removed if `keep_blank_lines` is false
132/// - Whitespace within lines is normalized to single spaces
133/// - Anchor markers like {#id} are removed iff `strip_anchors`
134fn normalize_plaintext(
135  plain_text: &str,
136  keep_blank_lines: bool,
137  strip_anchors: bool,
138) -> String {
139  let mut result = String::with_capacity(plain_text.len());
140  let mut line = String::new();
141  let mut prev_empty = false;
142  let mut has_output = false;
143
144  for raw_line in plain_text.lines() {
145    line.clear();
146
147    // Remove anchor markers {#id} and collapse whitespace.
148    for word in raw_line.split_whitespace().filter(|word| {
149      !(strip_anchors && word.starts_with("{#") && word.ends_with('}'))
150    }) {
151      if !line.is_empty() {
152        line.push(' ');
153      }
154      line.push_str(word);
155    }
156
157    if line.is_empty() {
158      if keep_blank_lines && !prev_empty && has_output {
159        result.push('\n');
160      }
161      prev_empty = true;
162    } else {
163      if has_output {
164        result.push('\n');
165      }
166      result.push_str(&line);
167      has_output = true;
168      prev_empty = false;
169    }
170  }
171
172  result.trim().to_string()
173}

175#[cfg(test)]

179  #[test]
180  fn content_to_plaintext_removes_anchor_markers() {
181    assert_eq!(
182      content_to_plaintext("# Title {#title}\n\nSome text."),
183      "Title\nSome text."
184    );
185  }
186
187  #[test]
188  fn html_to_plaintext_keeps_anchor_like_code() {
189    let html = "<p>Text</p><pre><code>{#literal-code-token}</code></pre>";
190    assert_eq!(html_to_plaintext(html), "Text\n{#literal-code-token}");
191  }
192
193  #[test]
194  fn html_to_plaintext_separates_blocks() {
195    let html = "<h1 id=\"a\">Title</h1><p>Some <em>text</em> and \
196                <code>code</code>.</p><ul><li>One</li><li>Two</li></ul>";
197    assert_eq!(
198      html_to_plaintext(html),
199      "Title\nSome text and code.\nOne\nTwo"
200    );
201  }
202
203  #[test]
204  fn html_to_plaintext_keeps_code_blocks_and_skips_scripts() {
205    let html = "<p>Before<br>after</p><pre><code>ls -la\ncd \
206                /</code></pre><script>alert(1)</script><style>p \
207                {}</style><p>End</p>";
208    assert_eq!(html_to_plaintext(html), "Before\nafter\nls -la\ncd /\nEnd");
209  }
