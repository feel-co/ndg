9/// A complete `nixosOptionsDoc` JSON document keyed by option name.
10pub type NixOptionsDocument = IndexMap<String, NixOptionDocument>;
11
12/// One option entry from `nixosOptionsDoc`.
13///
14/// Unknown fields remain accepted because `transformOptions` may add metadata.
15/// Known fields are typed so malformed input fails at the field that caused the
16/// problem instead of silently rendering an empty value.
17#[derive(Debug, Clone, Deserialize)]
18#[serde(rename_all = "camelCase")]
19pub struct NixOptionDocument {
20  /// Human-readable Nix option type.
21  #[serde(rename = "type")]
22  pub type_name: String,
23
24  /// Option description, either plain Markdown or a literal wrapper.
25  #[serde(default)]
26  pub description: Option<DocumentationText>,
27
28  /// Rendered default value.
29  #[serde(default, deserialize_with = "deserialize_documented_value")]
30  pub default: Option<DocumentedValue>,
31
32  /// Rendered example value.
33  #[serde(default, deserialize_with = "deserialize_documented_value")]
34  pub example: Option<DocumentedValue>,
35
36  /// Locations that declare the option.
37  #[serde(default)]
38  pub declarations: Vec<OptionLocation>,
39
40  /// Locations that define the option. This is an NDG-compatible extension.
41  #[serde(default)]
42  pub definitions: Vec<OptionLocation>,
43
44  /// Option attribute path as separate components.
45  pub loc: Vec<String>,
46
47  /// Whether the option is internal.
48  #[serde(default)]
49  pub internal: bool,
50
51  /// Whether the option is read-only.
52  #[serde(default)]
53  pub read_only: bool,
54
55  /// Visibility metadata retained by custom `transformOptions` functions.
56  #[serde(default)]
57  pub visible: Option<OptionVisibility>,
58
59  /// Legacy direct declaration URL.
60  #[serde(default, rename = "declarationURL")]
61  pub declaration_url: Option<String>,
62
63  /// Markdown links for packages related to this option.
64  #[serde(default)]
65  pub related_packages: Option<String>,
66}
67
68impl NixOptionDocument {
69  /// Return whether this entry should be treated as hidden or internal.
70  #[must_use]
71  pub fn is_hidden(&self) -> bool {
72    self.internal
73      || self
74        .visible
75        .as_ref()
76        .is_some_and(OptionVisibility::hides_option)
77  }
78}

80/// Markdown-bearing text used by descriptions and related package metadata.
81#[derive(Debug, Clone, Deserialize)]
82#[serde(untagged)]
83pub enum DocumentationText {
84  /// Plain Markdown text.
85  Plain(String),
86
87  /// Markdown wrapped by `lib.mdDoc`.
88  Markdown(MarkdownDocumentation),
89}
90
91impl DocumentationText {
92  /// Return the contained text.
93  #[must_use]
94  pub fn text(&self) -> &str {
95    match self {
96      Self::Plain(text)
97      | Self::Markdown(MarkdownDocumentation::Markdown { text }) => text,
98    }
99  }
100
101  /// Return whether the text came from an explicit `mdDoc` wrapper.
102  #[must_use]
103  pub const fn is_markdown_documentation(&self) -> bool {
104    matches!(self, Self::Markdown(_))
105  }
106}

108/// Structured Markdown documentation produced by `lib.mdDoc`.
109#[derive(Debug, Clone, Deserialize)]
110#[serde(
111  tag = "_type",
112  deny_unknown_fields,
113  expecting = "an mdDoc object with a string `text` field"
114)]
115pub enum MarkdownDocumentation {
116  /// Markdown documentation.
117  #[serde(rename = "mdDoc")]
118  Markdown {
119    /// Markdown contents.
120    text: String,
121  },
122}

124/// A rendered option value emitted by `nixosOptionsDoc`.
125#[derive(Debug, Clone, Deserialize, PartialEq, Eq)]
126#[serde(
127  tag = "_type",
128  deny_unknown_fields,
129  expecting = "a literalExpression or literalMD object with a string `text` \
130               field"
131)]
132pub enum DocumentedValue {
133  /// Verbatim Nix source rendered as a Nix code block.
134  #[serde(rename = "literalExpression")]
135  LiteralExpression {
136    /// Nix expression source.
137    text: String,
138  },
139
140  /// Markdown rendered as documentation content.
141  #[serde(rename = "literalMD")]
142  LiteralMarkdown {
143    /// Markdown contents.
144    text: String,
145  },
146}

148fn deserialize_documented_value<'de, D>(
149  deserializer: D,
150) -> Result<Option<DocumentedValue>, D::Error>
151where
152  D: Deserializer<'de>,
153{
154  DocumentedValue::deserialize(deserializer).map(Some)
155}

157/// Compare option locations using `nixos-render-docs` ordering.
158///
159/// Components beginning with `enable` sort first, followed by components
160/// beginning with `package`, then all other components alphabetically.
161#[must_use]
162pub fn compare_option_locs(left: &[String], right: &[String]) -> Ordering {
163  fn priority(component: &str) -> u8 {
164    if component.starts_with("enable") {
165      0
166    } else if component.starts_with("package") {
167      1
168    } else {
169      2
170    }
171  }
172
173  left
174    .iter()
175    .map(|component| (priority(component), component))
176    .cmp(
177      right
178        .iter()
179        .map(|component| (priority(component), component)),
180    )
181}

202/// Visibility forms accepted by the Nix module system.
203#[derive(Debug, Clone, Deserialize)]
204#[serde(untagged)]
205pub enum OptionVisibility {
206  /// Boolean visibility used by `nixosOptionsDoc` output.
207  Boolean(bool),
208
209  /// Raw module-system visibility mode retained by custom exporters.
210  Mode(VisibilityMode),
211}
212
213impl OptionVisibility {
214  fn hides_option(&self) -> bool {
215    match self {
216      Self::Boolean(visible) => !visible,
217      Self::Mode(VisibilityMode::Transparent) => true,
218      Self::Mode(VisibilityMode::Shallow) => false,
219    }
220  }
221}
222
223/// Named module-system visibility modes.
224#[derive(Debug, Clone, Deserialize)]
225#[serde(rename_all = "lowercase")]
226pub enum VisibilityMode {
227  /// Show this option while hiding its sub-options.
228  Shallow,
229
230  /// Hide this option while retaining visible sub-options.
231  Transparent,
232}

234/// Error returned while parsing an options JSON document.
235#[derive(Debug, Error)]
236pub enum OptionsParseError {
237  /// A value did not match the schema at the reported JSON path.
238  #[error("{0}")]
239  Schema(serde_path_to_error::Error<serde_json::Error>),
240
241  /// Extra non-whitespace data followed the JSON document.
242  #[error("{0}")]
243  Trailing(serde_json::Error),
244}
245
246/// Parse and validate a `nixosOptionsDoc` JSON document.
247///
248/// # Errors
249///
250/// Returns an error with the exact option and field path when a known field has
251/// the wrong type, a required field is absent, or the JSON syntax is invalid.
252pub fn parse_options_json(
253  input: &str,
254) -> Result<NixOptionsDocument, OptionsParseError> {
255  let mut deserializer = serde_json::Deserializer::from_str(input);
256  let document = serde_path_to_error::deserialize(&mut deserializer)
257    .map_err(OptionsParseError::Schema)?;
258  deserializer.end().map_err(OptionsParseError::Trailing)?;
259  Ok(document)
260}

268  #[test]
269  fn test_options_parser_accepts_current_nixos_options_doc_schema() {
270    let input = r#"{
271  "services.example.enable": {
272    "declarations": [
273      "nixos/modules/services/example.nix",
274      {"name": "module.nix", "url": "https://example.test/module.nix"}
275    ],
276    "default": {"_type": "literalExpression", "text": "false"},
277    "description": {"_type": "mdDoc", "text": "Enable **example**."},
278    "example": {"_type": "literalExpression", "text": "true"},
279    "loc": ["services", "example", "enable"],
280    "readOnly": false,
281    "relatedPackages": "- [`pkgs.example`](https://example.test)",
282    "type": "boolean"
283  }
284}"#;
285
286    let document = parse_options_json(input).expect("valid options JSON");
287    let option = document
288      .get("services.example.enable")
289      .expect("option entry");
290
291    assert_eq!(option.type_name, "boolean");
292    assert_eq!(option.declarations.len(), 2);
293    assert_eq!(option.loc, ["services", "example", "enable"]);
294    assert_eq!(
295      option.description.as_ref().map(DocumentationText::text),
296      Some("Enable **example**.")
297    );
298    assert_eq!(
299      option.related_packages.as_deref(),
300      Some("- [`pkgs.example`](https://example.test)")
301    );
302    assert_eq!(
303      option.default,
304      Some(DocumentedValue::LiteralExpression {
305        text: "false".to_string(),
306      })
307    );
308  }
