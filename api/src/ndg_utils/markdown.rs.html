24/// TOML frontmatter parsed from the `+++` delimited block at the start of a
25/// markdown document.
26///
27/// Frontmatter must appear at the very beginning of the file, wrapped in
28/// `+++` delimiters on their own lines. Any unrecognised keys are silently
29/// ignored by the TOML deserialiser.
30#[derive(Debug, Default, Clone, Serialize, Deserialize)]
31pub struct PageFrontmatter {
32  /// Overrides the page title extracted from the first H1 heading.
33  pub title:       Option<String>,
34  /// Per-page meta description. Injected as `<meta name="description">` and
35  /// as `og:description` where OG tags are enabled.
36  pub description: Option<String>,
37
38  /// Per-page author. Injected as `<meta name="author">`.
39  pub author: Option<String>,
40
41  /// Name of the template file to use for this page (e.g. `"custom"` or
42  /// `"custom.html"`). Falls back to the default template resolution if the
43  /// named file cannot be found.
44  pub template: Option<String>,
45
46  /// When `false`, suppresses the table of contents for this page.
47  pub toc: Option<bool>,
48
49  /// Per-page tags. Injected as `<meta name="keywords">` and override the
50  /// global `meta.tags.keywords` for this page.
51  pub tags: Option<Vec<String>>,
52
53  /// Arbitrary user-defined data. Accessible in templates as
54  /// `{{ page.extra.key }}`.
55  pub extra: Option<toml::Table>,
56}

58/// Strips TOML frontmatter from the start of a markdown document.
59///
60/// Frontmatter must begin at the very first byte of the file, delimited by
61/// `+++` on its own line. The closing `+++` must also be on its own line.
62/// Returns the parsed [`PageFrontmatter`] (if present and valid) and the
63/// remaining markdown content with the frontmatter block removed.
64///
65/// Malformed TOML inside a valid `+++` block is logged as a warning and
66/// treated as absent; the full original content is returned unchanged.
67fn strip_frontmatter(content: &str) -> (Option<PageFrontmatter>, String) {
68  const DELIM: &str = "+++";
69
70  // Normalize CRLF to LF so that Windows-style line endings are handled.
71  let lf_content;
72  let content: &str = if content.contains("\r\n") {
73    lf_content = content.replace("\r\n", "\n");
74    &lf_content
75  } else {
76    content
77  };
78
79  // The opening delimiter must be the very first thing in the file.
80  let Some(after_open) = content.strip_prefix(DELIM) else {
81    return (None, content.to_owned());
82  };
83  // Must be immediately followed by a newline.
84  let Some(after_open) = after_open.strip_prefix('\n') else {
85    return (None, content.to_owned());
86  };
87
88  // Find the closing delimiter.
89  let needle = format!("\n{DELIM}");
90  let Some(end_idx) = after_open.find(needle.as_str()) else {
91    return (None, content.to_owned());
92  };
93
94  let toml_str = &after_open[..end_idx];
95  let after_close = &after_open[end_idx + needle.len()..];
96  // Strip the single newline that follows the closing delimiter, if present.
97  let remaining = after_close.strip_prefix('\n').unwrap_or(after_close);
98
99  match toml::from_str::<PageFrontmatter>(toml_str) {
100    Ok(fm) => (Some(fm), remaining.to_owned()),
101    Err(e) => {
102      warn!("Failed to parse frontmatter: {e}");
103      (None, content.to_owned())
104    },
105  }
106}

143/// Processes all markdown files in the input directory and returns processed
144/// data.
145///
146/// This function renders all markdown files, handles custom output paths,
147/// and returns structured data for each file. Files that are included in other
148/// files are marked but still returned (caller decides whether to skip them).

150/// As a side effect, this function populates `config.included_files` with the

191fn process_markdown_files_impl(
192  config: &mut Config,
193  processor: Option<&MarkdownProcessor>,
194  cache_dir: Option<&Path>,
195) -> Result<Vec<ProcessedMarkdown>> {
196  if let Some(ref input_dir) = config.input_dir {
197    info!("Input directory: {}", input_dir.display());
198    let mut files = collect_markdown_files(input_dir);
199    files.sort();
200    info!("Found {} markdown files", files.len());
201
202    // Map: output html path -> output entry
203    #[expect(clippy::items_after_statements, reason = "Local type alias")]
204    type OutputEntry = (
205      String,
206      Vec<Header>,
207      Option<String>,
208      PathBuf,
209      bool,
210      Option<PageFrontmatter>,
211    );
212    let mut output_map: FxHashMap<String, OutputEntry> = FxHashMap::default();
213
214    // Track custom outputs keyed by included file path
215    let mut pending_custom_outputs: FxHashMap<PathBuf, Vec<String>> =
216      FxHashMap::default();
217
218    // Track generated output paths for included files with html:into-file.
219    let mut included_output_files: FxHashMap<PathBuf, PathBuf> =
220      FxHashMap::default();
221
222    // Track all included files across all markdown files
223    let mut all_included_files: FxHashMap<PathBuf, PathBuf> =
224      FxHashMap::default();
225
226    // Use provided processor or create a new one
227    let base_processor = processor
228      .cloned()
229      .map_or_else(|| create_processor(config, None), Ok)?;
230
231    let cache = cache_dir.map(|dir| (dir, processor_digest(&base_processor)));
232    let dependency_digests = DependencyDigests::default();
233    let progress = ProgressBar::new(files.len() as u64);
234    #[expect(
235      clippy::expect_used,
236      reason = "template string is a compile-time constant"
237    )]
238    let style = ProgressStyle::default_bar()
239      .template(
240        "{spinner:.green} [{elapsed_precise}] [{bar:40.cyan/blue}] \
241         {pos}/{len} ({eta}) {msg}",
242      )
243      .expect("valid progress bar template string")
244      .progress_chars("#>-");
245    progress.set_style(style);
246    progress.set_message("Processing markdown");
247    progress.enable_steady_tick(std::time::Duration::from_millis(100));
248
249    let rendered: Result<Vec<_>> = files
250      .par_iter()
251      .progress_with(progress.clone())
252      .map(|file_path| {
253        let raw_content =
254          fs::read_to_string(file_path).wrap_err_with(|| {
255            format!("Failed to read markdown file: {}", file_path.display())
256          })?;
257        let base_dir = file_path.parent().unwrap_or(input_dir.as_path());
258        let source_digest = digest(&raw_content);
259        if let Some((dir, processor_digest)) = &cache
260          && let Some(entry) = read_cache(
261            dir,
262            file_path,
263            base_dir,
264            &source_digest,
265            processor_digest,
266            &dependency_digests,
267          )
268        {
269          return Ok((entry.frontmatter, entry.result));
270        }
271
272        let (frontmatter, content) = strip_frontmatter(&raw_content);
273        // Render without validation; duplicate-anchor policy (including
274        // `[sidebar.toc]` exclusions) is applied after cache lookup so cached
275        // and fresh renders behave identically.
276        let result = base_processor
277          .clone()
278          .with_base_dir(base_dir)
279          .render(&content);
280        if let Some((dir, processor_digest)) = &cache {
281          write_cache(
282            dir,
283            file_path,
284            base_dir,
285            &source_digest,
286            processor_digest,
287            &result,
288            frontmatter.as_ref(),
289          );
290        }
291        Ok((frontmatter, result))
292      })
293      .collect();
294    progress.finish_with_message("Markdown processing complete");
295
296    let mut rendered = rendered?;
297    // Apply duplicate-anchor policy per file after cache lookup so cached and
298    // fresh renders behave identically.
299    for (file_path, (_, result)) in files.iter().zip(rendered.iter_mut()) {
300      apply_anchor_policy(config, file_path, result)?;
301    }
302    let rendered = rendered;
303    for (file_path, (_, result)) in files.iter().zip(&rendered) {
304      let base_dir = file_path.parent().unwrap_or(input_dir.as_path());
305
306      // Track custom outputs for included files
307      for inc in &result.included_files {
308        if let Some(ref custom_output) = inc.custom_output {
309          let inc_path = base_dir.join(&inc.path);
310          if let Ok(inc_rel) = inc_path.strip_prefix(input_dir) {
311            let custom_output = Path::new(custom_output);
312            crate::output::output_path(&config.output_dir, custom_output)
313              .wrap_err_with(|| {
314                format!(
315                  "Invalid html:into-file output path: {}",
316                  custom_output.display()
317                )
318              })?;
319            let normalized_output = custom_output.to_path_buf();
320            pending_custom_outputs
321              .entry(inc_rel.to_path_buf())
322              .or_default()
323              .push(normalized_output.to_string_lossy().to_string());
324            included_output_files
325              .insert(inc_rel.to_path_buf(), normalized_output);
326          }
327        }
328      }
329
330      // Collect all included files for this markdown file
331      for inc in &result.included_files {
332        let inc_path = base_dir.join(&inc.path);
333        if let Ok(inc_rel) = inc_path.strip_prefix(input_dir) {
334          let Ok(includer_rel) = file_path.strip_prefix(input_dir) else {
335            continue;
336          };
337
338          all_included_files
339            .entry(inc_rel.to_path_buf())
340            .and_modify(|old| {
341              // For deterministic output
342              if includer_rel < old.as_path() {
343                *old = includer_rel.to_path_buf();
344              }
345            })
346            .or_insert_with(|| includer_rel.to_owned());
347        }
348      }
349    }
350
351    // Second pass: build output entries after resolving include relationships.
352    for (file_path, (frontmatter, result)) in files.into_iter().zip(rendered) {
353      let rel_path = file_path.strip_prefix(input_dir).wrap_err_with(|| {
354        format!(
355          "Failed to determine relative path for {}",
356          file_path.display()
357        )
358      })?;
359      let mut output_path = rel_path.to_path_buf();
360      output_path.set_extension("html");
361
362      // If index.use_readme is enabled and this is README.md, output to
363      // index.html
364      if config.use_readme_as_homepage() {
365        let file_name = rel_path
366          .file_name()
367          .and_then(|n| n.to_str())
368          .unwrap_or("")
369          .to_ascii_lowercase();
370        if file_name == "readme.md" {
371          output_path.set_file_name("index.html");
372        }
373      }
374
375      let output_path_str = output_path.to_string_lossy().to_string();
376
377      // Check if this file is included in another file
378      let is_included = all_included_files.contains_key(rel_path);
379
380      // Get any pending custom outputs for this file
381      let custom_outputs =
382        pending_custom_outputs.remove(rel_path).unwrap_or_default();
383
384      // If this file has custom outputs via html:into-file, write to those
385      // locations
386      if custom_outputs.is_empty() {
387        output_map.insert(
388          output_path_str,
389          (
390            result.html,
391            result.headers,
392            result.title,
393            file_path,
394            is_included,
395            frontmatter,
396          ),
397        );
398      } else {
399        for custom in custom_outputs {
400          output_map.insert(
401            custom,
402            (
403              result.html.clone(),
404              result.headers.clone(),
405              result.title.clone(),
406              file_path.clone(),
407              false,
408              frontmatter.clone(),
409            ),
410          );
411        }
412      }
413    }
414
415    // Populate config.included_files so caller can use it
416    config.included_files.clone_from(&all_included_files);
417    config
418      .included_output_files
419      .clone_from(&included_output_files);
420
421    // Convert output_map to Vec<ProcessedMarkdown>
422    let mut results: Vec<ProcessedMarkdown> = output_map
423      .into_iter()
424      .map(
425        |(
426          output_path,
427          (html_content, headers, title, source_path, is_included, frontmatter),
428        )| {
429          ProcessedMarkdown {
430            html_content,
431            headers,
432            title,
433            source_path,
434            output_path,
435            is_included,
436            frontmatter,
437          }
438        },
439      )
440      .collect();
441
442    // Sort for deterministic output
443    results.sort_by(|a, b| a.output_path.cmp(&b.output_path));
444
445    Ok(results)
446  } else {
447    info!("No input directory provided, skipping markdown processing");
448    Ok(Vec::new())
449  }
450}

452/// Validate or deduplicate heading anchors for a single rendered page.
453///
454/// The configured `[anchor] on_duplicate` policy decides whether duplicates
455/// fail the build (`error`), log a warning (`warn`), or are made unique with
456/// `-1`, `-2`, ... suffixes (`deduplicate`).
457fn apply_anchor_policy(
458  config: &Config,
459  file_path: &Path,
460  result: &mut ndg_commonmark::MarkdownResult,
461) -> Result<()> {
462  let policy = config.duplicate_anchor_policy();
463  match policy {
464    DuplicateAnchorPolicy::Error => {
465      validate_rendered_anchor_ids(&result.headers, &result.html).map_err(|e| {
466        color_eyre::eyre::eyre!(
467          "Duplicate anchor IDs in {}:\n{e}",
468          file_path.display()
469        )
470      })
471    },
472    DuplicateAnchorPolicy::Warn => {
473      if let Err(e) =
474        validate_rendered_anchor_ids(&result.headers, &result.html)
475      {
476        warn!("Duplicate anchor IDs in {}:\n{e}", file_path.display());
477      }
478      Ok(())
479    },
480    DuplicateAnchorPolicy::Deduplicate => {
481      let renames =
482        deduplicate_anchor_ids(&mut result.headers, &mut result.html);
483      for (old, new) in &renames {
484        info!(
485          "Deduplicated anchor ID '{old}' -> '{new}' in {}",
486          file_path.display()
487        );
488      }
489      Ok(())
490    },
491  }
492}

494fn digest(content: impl AsRef<[u8]>) -> String {
495  blake3::hash(content.as_ref()).to_hex().to_string()
496}
497
498fn digest_file(path: &Path) -> Option<String> {
499  fs::read(path).ok().map(digest)
500}
501
502fn digest_tree(path: Option<&str>) -> String {
503  let Some(path) = path else {
504    return String::new();
505  };
506  let path = Path::new(path);
507  if path.is_file() {
508    return digest_file(path).unwrap_or_default();
509  }
510  let mut files: Vec<_> = walkdir::WalkDir::new(path)
511    .into_iter()
512    .filter_map(Result::ok)
513    .filter(|entry| !entry.file_type().is_dir())
514    .map(walkdir::DirEntry::into_path)
515    .collect();
516  files.sort();
517  let mut hasher = blake3::Hasher::new();
518  for file in files {
519    hasher.update(file.to_string_lossy().as_bytes());
520    if let Ok(content) = fs::read(&file) {
521      hasher.update(&content);
522    }
523  }
524  hasher.finalize().to_hex().to_string()
525}

531fn processor_digest_for_version(
532  processor: &MarkdownProcessor,
533  version: &str,
534) -> String {
535  let options = processor.options();
536  digest(format!(
537    "{MARKDOWN_CACHE_SCHEMA}|{version}|{:?}|{}|{}|{}|{}",
538    options,
539    digest_tree(options.syntax_queries_path.as_deref()),
540    cfg!(feature = "commonmark-ndg"),
541    cfg!(feature = "commonmark-nixpkgs"),
542    cfg!(feature = "commonmark-nixpkgs") || cfg!(feature = "commonmark-ndg"),
543  ))
544}

553fn read_cache(
554  cache_dir: &Path,
555  source: &Path,
556  base_dir: &Path,
557  source_digest: &str,
558  processor_digest: &str,
559  dependency_digests: &DependencyDigests,
560) -> Option<MarkdownCacheEntry> {
561  let entry: MarkdownCacheEntry =
562    serde_json::from_slice(&fs::read(cache_path(cache_dir, source)).ok()?)
563      .ok()?;
564  if entry.schema != MARKDOWN_CACHE_SCHEMA
565    || entry.source_digest != source_digest
566    || entry.processor_digest != processor_digest
567  {
568    return None;
569  }
570  entry
571    .dependencies
572    .iter()
573    .all(|(path, expected)| {
574      dependency_matches(
575        &base_dir.join(path),
576        expected.as_deref(),
577        dependency_digests,
578      )
579    })
580    .then_some(entry)
581}
582
583fn dependency_matches(
584  path: &Path,
585  expected: Option<&str>,
586  digests: &DependencyDigests,
587) -> bool {
588  // ponytail: first reads are serialized; shard if unique includes dominate.
589  let Ok(mut digests) = digests.lock() else {
590    return false;
591  };
592  digests
593    .entry(path.to_owned())
594    .or_insert_with(|| digest_file(path))
595    .as_deref()
596    == expected
597}
598
599fn write_cache(
600  cache_dir: &Path,
601  source: &Path,
602  base_dir: &Path,
603  source_digest: &str,
604  processor_digest: &str,
605  result: &ndg_commonmark::MarkdownResult,
606  frontmatter: Option<&PageFrontmatter>,
607) {
608  let dependencies = result
609    .included_files
610    .iter()
611    .map(|include| {
612      let path = PathBuf::from(&include.path);
613      let digest = digest_file(&base_dir.join(&path));
614      (path, digest)
615    })
616    .collect();
617  let entry = MarkdownCacheEntryRef {
618    schema: MARKDOWN_CACHE_SCHEMA,
619    source_digest,
620    dependencies,
621    processor_digest,
622    result,
623    frontmatter,
624  };
625  let Ok(data) = serde_json::to_vec(&entry) else {
626    return;
627  };
628  if fs::create_dir_all(cache_dir).is_err() {
629    return;
630  }
631  let path = cache_path(cache_dir, source);
632  let temp = path.with_extension(format!("{}.tmp", std::process::id()));
633  if fs::write(&temp, data).is_ok() {
634    let _ = fs::rename(temp, path);
635  }
636}

638/// Creates a markdown processor from the NDG configuration.
639///
640/// # Arguments
641/// * `config` - The loaded configuration for documentation generation.
642/// * `valid_options` - Optional set of valid option names for validation.
643///
644/// # Errors
645///
646/// Returns an error when configured manpage URL mappings cannot be read or
647/// parsed.
648#[expect(
649  clippy::implicit_hasher,
650  reason = "Standard FxHashSet sufficient for this use case"
651)]
652pub fn create_processor(
653  config: &Config,
654  valid_options: Option<FxHashSet<String>>,
655) -> Result<MarkdownProcessor> {
656  let tab_style = match config.tab_style.as_str() {
657    "warn" => TabStyle::Warn,
658    "normalize" => TabStyle::Normalize,
659    _ => TabStyle::None,
660  };
661
662  let mut builder = MarkdownOptionsBuilder::new()
663    .gfm(true)
664    .highlight_code(config.highlight_code)
665    .extensions(
666      config
667        .markdown
668        .as_ref()
669        .map_or_else(Vec::new, |markdown| markdown.extensions.clone()),
670    )
671    .tab_style(tab_style);
672
673  if let Some(mappings_path) = &config.manpage_urls_path {
674    let mappings = ndg_commonmark::utils::load_manpage_urls(mappings_path)
675      .wrap_err_with(|| {
676        format!(
677          "Failed to load manpage URL mappings from {}",
678          mappings_path.display()
679        )
680      })?;
681    builder = builder.manpage_urls(Some(mappings));
682  }
683
684  if let Some(queries_path) = &config.syntax_queries_path {
685    builder = builder
686      .syntax_queries_path(Some(queries_path.to_string_lossy().to_string()));
687  }
688
689  if let Some(options) = valid_options {
690    builder = builder.valid_options(Some(options));
691  }
692
693  Ok(MarkdownProcessor::new(builder.build()))
694}

696/// Extracts the page title from a markdown file.
697///
698/// This attempts to extract the first heading as the title, falling back to the
699/// file name if no heading is found or the file cannot be read.
700///
701/// # Arguments
702///
703/// * `file_path` - The path to the markdown file.
704/// * `html_path` - The corresponding HTML output path (used for fallback).
705///
706/// # Returns
707///
708/// The extracted title or the file stem as a fallback.
709#[must_use]
710pub fn extract_page_title(file_path: &Path, html_path: &Path) -> String {
711  let default_title = html_path
712    .file_stem()
713    .unwrap_or_default()
714    .to_string_lossy()
715    .to_string();
716
717  match fs::read_to_string(file_path) {
718    Ok(content) => {
719      ndg_commonmark::utils::extract_markdown_title_and_id(&content)
720        .map_or(default_title, |(title, _)| title)
721    },
722    Err(e) => {
723      log::warn!(
724        "Failed to read {} for title extraction: {e}",
725        file_path.display()
726      );
727      default_title
728    },
729  }
730}

764    let source = base.join("page.md");
765    let include = base.join("part.md");

768    fs::write(&include, "included v1").unwrap();

770    let result = MarkdownResult {
771      html:           String::new(),
772      headers:        Vec::new(),
773      title:          None,
774      included_files: vec![IncludedFile {
775        path:          "part.md".to_string(),
776        custom_output: None,
777      }],
778    };
779    let source_digest = digest("# Page");
780    write_cache(
781      &cache,
782      &source,
783      &base,
784      &source_digest,
785      "processor-a",
786      &result,
787      None,
788    );
789
790    assert!(
791      read_cache(&cache, &source, &base, &source_digest, "processor-a")
792        .is_some(),

794    );
795    assert!(
796      read_cache(&cache, &source, &base, &digest("# Changed"), "processor-a")
797        .is_none(),
798      "changed Markdown should invalidate the cache"
799    );
800
801    fs::write(&include, "included v2").unwrap();
802    assert!(
803      read_cache(&cache, &source, &base, &source_digest, "processor-a")
804        .is_none(),
805      "changed includes should invalidate the cache"
806    );
807
808    fs::write(&include, "included v1").unwrap();
809    assert!(
810      read_cache(&cache, &source, &base, &source_digest, "processor-b")
811        .is_none(),

1008  /// Replicates <https://github.com/feel-co/ndg/issues/272#issuecomment-5476940549>:
1009  /// repeated `Inputs`/`Type`/`Examples` sections must fail by default even
1010  /// when excluded from the TOC, and follow explicit duplicate policies.
1011  #[test]
1012  fn issue_272_duplicate_sections_end_to_end() {
1013    use ndg_config::anchor::AnchorConfig;
1014
1015    const DOC: &str = "# Module\n\n### Inputs\n\nFirst.\n\n### \
1016                       Type\n\nSecond.\n\n### Examples\n\nThird.\n\n### \
1017                       Inputs\n\nFourth.\n\n### Type\n\nFifth.\n\n### \
1018                       Examples\n\nSixth.\n";
1019
1020    fn run(
1021      policy: Option<DuplicateAnchorPolicy>,
1022      exclude: bool,
1023    ) -> Result<usize> {
1024      let temp = TempDir::new().unwrap();
1025      let input = temp.path().join("docs");
1026      fs::create_dir_all(&input).unwrap();
1027      fs::write(input.join("module-schema.md"), DOC).unwrap();
1028      let mut config = Config {
1029        input_dir: Some(input),
1030        anchor: policy.map(|on_duplicate| {
1031          AnchorConfig {
1032            on_duplicate,
1033            ..Default::default()
1034          }
1035        }),
1036        ..Config::default()
1037      };
1038      if exclude {
1039        use ndg_config::sidebar::SidebarTocConfig;
1040        let mut sidebar = ndg_config::sidebar::SidebarConfig::default();
1041        sidebar.toc = SidebarTocConfig {
1042          exclude: vec![ndg_config::sidebar::TitleMatch {
1043            exact:          None,
1044            regex:          Some("^(Inputs|Type|Examples)$".to_string()),
1045            compiled_regex: None,
1046          }],
1047        };
1048        sidebar.validate().unwrap();
1049        config.sidebar = Some(sidebar);
1050      }
1051      process_markdown_files(&mut config, None)
1052        .map(|pages| pages.iter().map(|page| page.headers.len()).sum())
1053    }
1054
1055    assert!(run(None, false).is_err(), "duplicates must fail by default");
1056    assert!(
1057      run(None, true).is_err(),
1058      "TOC exclusions do not alter anchors"
1059    );
1060    assert!(run(Some(DuplicateAnchorPolicy::Warn), true).is_ok());
1061    let pages = {
1062      let temp = TempDir::new().unwrap();
1063      let input = temp.path().join("docs");
1064      fs::create_dir_all(&input).unwrap();
1065      fs::write(input.join("module-schema.md"), DOC).unwrap();
1066      let mut config = Config {
1067        input_dir: Some(input),
1068        anchor: Some(AnchorConfig {
1069          on_duplicate: DuplicateAnchorPolicy::Deduplicate,
1070          ..Default::default()
1071        }),
1072        ..Config::default()
1073      };
1074      process_markdown_files(&mut config, None).unwrap()
1075    };
1076    let ids: Vec<&str> = pages[0]
1077      .headers
1078      .iter()
1079      .map(|header| header.id.as_str())
1080      .collect();
1081    assert_eq!(ids, vec![
1082      "module",
1083      "inputs",
1084      "type",
1085      "examples",
1086      "inputs-1",
1087      "type-1",
1088      "examples-1"
1089    ]);
1090    validate_rendered_anchor_ids(&pages[0].headers, &pages[0].html_content)
1091      .unwrap();
1092  }
